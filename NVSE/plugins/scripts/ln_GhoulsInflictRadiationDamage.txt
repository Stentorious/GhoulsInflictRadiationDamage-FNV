	return
endif

; Load INI settings

; Load factions
; Mutated

Goo1.AuxVarSetFlt "*GIRD_Fact" (eval GetINIFloat_Cached "Creature:bCentaur" "Stentorious\GhoulsInflictRadiationDamage.ini" 1) 1

CloseFileSO "Stentorious\GhoulsInflictRadiationDamage.ini" 0

; Remove reaver rad damage
if GetNthEffectBase DLC03FeralGhoulPoison 0 == DamageRadiationLevel
	RemoveNthEffect DLC03FeralGhoulPoison 0
endif
if eval IsModLoaded "PointLookout.esm"
	ref rEffect = EditorIDToFormID "DLC04FeralGhoulPoison"
	if eval IsFormValid rEffect && GetNthEffectBase rEffect 0 == DamageRadiationLevel
		RemoveNthEffect rEffect 0
	endif
endif

; Reuse GoldenGeckoRadiation effect
AddNewEffect GoldenGeckoRadiation DamageRadiationLevel 0 0 0
int iSize = GetNumEffects GoldenGeckoRadiation
while (iSize -= 1) > -1
	RemoveNthEffect GoldenGeckoRadiation iSize
loop
LNSetName "Radiation Damage" GoldenGeckoRadiation

; Load keywords
array_var aForms = GetKeywordForms "InflictRadDamage"
if eval (iSize = ar_Size aForms) > 0
	while (iSize -= 1) > -1
		if eval IsBaseForm (ref rTemp = aForms[iSize]) && (GetType rTemp == 8 || GetType rTemp == 43)
			AuxVarSetFlt "*InflictRadDamage" 1 0 rTemp
		endif
	loop
endif
aForms = Ar_Null

; Update radiation dealing factions
Call (CompileScript "GhoulsInflictRadiationDamage\UpdateForms.gek")

; Event handlers
SetEventHandler "OnHitWith" (CompileScript "GhoulsInflictRadiationDamage\OnHitWith.gek") "first"::Player "second"::Fists
SetEventHandler "OnHitWith" (CompileScript "GhoulsInflictRadiationDamage\OnHitWith.gek") "first"::Player "second"::Bloatfly1HP
SetEventHandler "OnHitWith" (CompileScript "GhoulsInflictRadiationDamage\OnHitWith.gek") "first"::Player "second"::SporePlantSpitMissile
SetEventHandler "OnHitWith" (CompileScript "GhoulsInflictRadiationDamage\OnHitWith.gek") "first"::Player "second"::NVDLC02WeapSporePlantSpitMissile
SetEventHandler "OnHitWith" (CompileScript "GhoulsInflictRadiationDamage\OnHitWith.gek") "first"::Player "second"::NVDLC02WeapSporePlantSpitMissileLvl2

; MCM update event
SetEventHandler "MCMExtUpdate" (begin function {array_var aTemp}
	if eval (Ar_Find "CreatureRadiationDamage" aTemp) != Ar_BadNumericIndex
		Call (CompileScript "GhoulsInflictRadiationDamage\UpdateForms.gek")
	endif
	aTemp = ar_Null
end)

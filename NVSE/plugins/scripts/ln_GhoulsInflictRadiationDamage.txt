; Ghouls Inflict Radiation Damage - Stentorious

; On Game restart
if eval Goo1.AuxVarGetFlt "*GIRD_Init" 0
	return
endif

; Check for requirements
if eval GetNVSEVersion > 5 && GetNVSERevision > 2 && GetNVSEBeta > 9
else
	MessageBoxEx "Ghouls Inflict Radiation missing requirement!%rInstall xNVSE 6.3.10+."
	return
endif
if eval GetPluginVersion "JohnnyGuitarNVSE" < 390
	MessageBoxEx "Ghouls Inflict Radiation missing requirement!%rInstall latest JohnnyGuitar NVSE plugin."
	return
endif

	MessageBoxEx "Ghouls Inflict Radiation missing requirement!%rInstall latest ShowOff NVSE plugin."
	return
endif

		RemoveNthEffect DLC03FeralGhoulPoison 0
	endif
	if eval IsModLoaded "PointLookout.esm"
		ref rEffect = EditorIDToFormID "DLC04FeralGhoulPoison"
		if eval IsFormValid rEffect && GetNthEffectBase rEffect 0 == DamageRadiationLevel
			RemoveNthEffect rEffect 0
		endif
	endif
endif

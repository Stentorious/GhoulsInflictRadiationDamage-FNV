; Clone radiation effect
SetEventHandlerAlt "GIRDClone" (begin Function {}

	if eval GetType (Goo1.AuxVarGetRef "*GIRDEffect") != 20
		ref stR1 = TempCloneForm GoldenGeckoRadiation 1
		stR1.LNSetName "Radiation Damage"
		SetNthEffectTraitNumeric stR1 0 2 0
		Goo1.AuxVarSetRef "*GIRDEffect" stR1
	endif

end)

; On load
DispatchEventAlt "GIRDClone"

; On unarmed attack
SetEventHandlerAlt "OnHitWith" (begin Function {ref stL1, ref stL2}
	ref stR1 = stL1.GetHitAttacker
	if IsFormValid stR1
		array_var aIter
		array_var stA1 = GetBaseFactions stR1
		array_var aResult = &(ar_FindWhere stA1 ({aIter} => HasKeyword (*aIter) "InflictRadDamage" ))
		if eval ((TypeOf *aResult) != "Number" || (*aResult != 0) || (HasKeyword (stR1.gbf) "InflictRadDamage"))
			DispatchEventAlt "GIRDCI" stR1 (stR1.GetAttackDamage)
		endif
		stA1 = aIter = aResult = ar_Null
	endif
end) 1::Player 2::Fists
